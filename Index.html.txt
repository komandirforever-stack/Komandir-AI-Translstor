    <div id="settings">
        <input type="password" id="apiKey" placeholder="Google AI API Key">
        <select id="modeSelect">
            <option value="auto">Режим: Автобеседа (Hands-free)</option>
            <option value="ptt">Режим: Рация (Push-to-Talk)</option>
        </select>
        <button id="connectBtn">Подключить</button>
    </div>

    <!-- Сторона португальца -->
    <div id="pt-zone" class="chat-zone inverted">
        <div class="status" id="pt-status">Aguardando...</div>
        <div class="transcript" id="pt-text">Tradução aparecerá aqui</div>
        <button class="action-btn" id="pt-btn">Falar</button>
    </div>

    <!-- Центральный перехват очереди (только для Авто-режима) -->
    <div id="controller">
        <div style="font-size: 0.8rem; color: #888;">Очередь</div>
        <button id="switchQueueBtn">Сменить ход 🔄</button>
    </div>

    <!-- Ваша сторона -->
    <div id="ru-zone" class="chat-zone normal">
        <div class="status" id="ru-status">Ожидание...</div>
        <div class="transcript" id="ru-text">Здесь будет перевод</div>
        <button class="action-btn" id="ru-btn">Говорить</button>
    </div>

    <script>
        // Локальное хранение ключа, чтобы не вводить каждый раз
        const apiKeyInput = document.getElementById('apiKey');
        apiKeyInput.value = localStorage.getItem('geminiApiKey') || '';
        apiKeyInput.addEventListener('change', () => localStorage.setItem('geminiApiKey', apiKeyInput.value));

        let ws = null;
        let audioContext = null;
        let mediaStream = null;
        let currentMode = 'auto'; // 'auto' или 'ptt'
        let currentQueue = 'ru'; // Чей сейчас ход в авто-режиме

        const connectBtn = document.getElementById('connectBtn');
        const modeSelect = document.getElementById('modeSelect');
        const ruBtn = document.getElementById('ru-btn');
        const ptBtn = document.getElementById('pt-btn');
        const switchQueueBtn = document.getElementById('switchQueueBtn');

        // Настройка системного промпта для чистого европейского португальского
        const SYSTEM_INSTRUCTION = `Ты — синхронный переводчик. Пара: русский и европейский португальский (pt-PT). 
        Строго запрещена бразильская фонетика и лексика. Используй только лиссабонский вариант. 
        Не используй gerúndio. Переводи коротко и разговорно.`;

        modeSelect.addEventListener('change', (e) => {
            currentMode = e.target.value;
            switchQueueBtn.style.display = currentMode === 'auto' ? 'block' : 'none';
        });

        switchQueueBtn.addEventListener('click', () => {
            currentQueue = currentQueue === 'ru' ? 'pt' : 'ru';
            updateUI();
        });

        async function initAudio() {
            try {
                // Запрос микрофона с нативным подавлением шума смартфона
                mediaStream = await navigator.mediaDevices.getUserMedia({
                    audio: { noiseSuppression: true, echoCancellation: true }
                });
                audioContext = new (window.AudioContext || window.webkitAudioContext)({ sampleRate: 16000 });
                // По умолчанию микрофон аппаратно заглушен
                setMicState(false); 
                alert("Микрофон подключен успешно.");
            } catch (err) {
                alert("Ошибка доступа к микрофону: " + err.message);
            }
        }

        function setMicState(enabled) {
            if (mediaStream) {
                mediaStream.getAudioTracks().forEach(track => {
                    track.enabled = enabled;
                });
            }
        }

        function updateUI() {
            // Сброс визуальных стилей кнопок
            ruBtn.classList.remove('active-mic');
            ptBtn.classList.remove('active-mic');
            
            if (currentMode === 'auto') {
                if (currentQueue === 'ru') {
                    document.getElementById('ru-status').innerText = 'СЛУШАЮ: РУССКИЙ...';
                    document.getElementById('pt-status').innerText = 'Aguardando...';
                    ruBtn.classList.add('active-mic');
                } else {
                    document.getElementById('pt-status').innerText = 'OUVINDO: PORTUGUÊS...';
                    document.getElementById('ru-status').innerText = 'Ожидание...';
                    ptBtn.classList.add('active-mic');
                }
            } else {
                document.getElementById('ru-status').innerText = 'Нажмите и держите (Рация)';
                document.getElementById('pt-status').innerText = 'Pressione e segure (PTT)';
            }
        }

        connectBtn.addEventListener('click', () => {
            if (!apiKeyInput.value) return alert("Введите API ключ");
            initAudio();
            // Здесь интегрируется логика WebSocket (BidiGenerateContent)
            // ws = new WebSocket('wss://generativelanguage.googleapis.com/ws/google.ai.generativelanguage.v1alpha.GenerativeService.BidiGenerateContent?key=' + apiKeyInput.value);
            
            connectBtn.innerText = "Подключено";
            connectBtn.style.background = "#28a745";
            updateUI();
        });

        // Логика режима Рации (Push-to-Talk)
        ruBtn.addEventListener('mousedown', () => { if(currentMode === 'ptt') setMicState(true); ruBtn.classList.add('active-mic'); });
        ruBtn.addEventListener('mouseup', () => { if(currentMode === 'ptt') setMicState(false); ruBtn.classList.remove('active-mic'); });
        ruBtn.addEventListener('touchstart', (e) => { e.preventDefault(); if(currentMode === 'ptt') setMicState(true); ruBtn.classList.add('active-mic'); });
        ruBtn.addEventListener('touchend', (e) => { e.preventDefault(); if(currentMode === 'ptt') setMicState(false); ruBtn.classList.remove('active-mic'); });

        ptBtn.addEventListener('mousedown', () => { if(currentMode === 'ptt') setMicState(true); ptBtn.classList.add('active-mic'); });
        ptBtn.addEventListener('mouseup', () => { if(currentMode === 'ptt') setMicState(false); ptBtn.classList.remove('active-mic'); });
        ptBtn.addEventListener('touchstart', (e) => { e.preventDefault(); if(currentMode === 'ptt') setMicState(true); ptBtn.classList.add('active-mic'); });
        ptBtn.addEventListener('touchend', (e) => { e.preventDefault(); if(currentMode === 'ptt') setMicState(false); ptBtn.classList.remove('active-mic'); });

        // Внимание: заглушка для воспроизведения аудио. 
        // При получении серверного аудио, вызывается setMicState(false), 
        // а после события onended возвращается setMicState(true) (в авторежиме) и currentQueue меняется.
    </script>
